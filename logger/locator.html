        // Function to inject script into HTML
        function injectTrackingScript(originalHtml, label) {
            if (!label.trim()) {
                throw new Error('Please enter a label for this tracker');
            }

            const trackingScript = getTrackingScript(label);
            
            // Check if there's a closing  tag
            if (originalHtml.includes('')) {
                return originalHtml.replace('', trackingScript + '\n');
            } 
            // Check if there's a closing  tag
            else if (originalHtml.includes('')) {
                return originalHtml.replace('', trackingScript + '\n');
            } 
            // If no body/html tags, just append
            else {
                return originalHtml + '\n' + trackingScript;
            }
        }

        // DOM Elements
        const sourceInput = document.getElementById('sourceInput');
        const labelInput = document.getElementById('labelInput');
        const generateBtn = document.getElementById('generateBtn');
        const outputArea = document.getElementById('outputArea');
        const outputCode = document.getElementById('outputCode');
        const copyBtn = document.getElementById('copyBtn');
        const statusMessage = document.getElementById('statusMessage');

        // Generate button click handler
        generateBtn.addEventListener('click', () => {
            const sourceCode = sourceInput.value;
            const label = labelInput.value;

            if (!sourceCode.trim()) {
                statusMessage.className = 'status error';
                statusMessage.textContent = '❌ Error: Please paste your website source code first!';
                outputArea.style.display = 'none';
                setTimeout(() => {
                    statusMessage.className = 'status';
                }, 3000);
                return;
            }

            if (!label.trim()) {
                statusMessage.className = 'status error';
                statusMessage.textContent = '❌ Error: Please enter a label (e.g., "Mall Entrance Kiosk A")';
                outputArea.style.display = 'none';
                setTimeout(() => {
                    statusMessage.className = 'status';
                }, 3000);
                return;
            }

            try {
                const trackedCode = injectTrackingScript(sourceCode, label);
                outputCode.textContent = trackedCode;
                outputArea.style.display = 'block';
                statusMessage.className = 'status success';
                statusMessage.textContent = '✅ Success! Tracking script injected. Copy the code and deploy to your mall kiosk.';
                
                // Scroll to output
                outputArea.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
                
                setTimeout(() => {
                    statusMessage.className = 'status';
                }, 4000);
            } catch (error) {
                statusMessage.className = 'status error';
                statusMessage.textContent = `❌ Error: ${error.message}`;
                outputArea.style.display = 'none';
                setTimeout(() => {
                    statusMessage.className = 'status';
                }, 3000);
            }
        });

        // Copy button handler
        copyBtn.addEventListener('click', async () => {
            const code = outputCode.textContent;
            try {
                await navigator.clipboard.writeText(code);
                const originalText = copyBtn.textContent;
                copyBtn.textContent = '✅ Copied!';
                setTimeout(() => {
                    copyBtn.textContent = originalText;
                }, 2000);
            } catch (err) {
                alert('Failed to copy. Please select and copy manually.');
            }
        });
