    // Robust HTML injection (preserve structure, insert before  or fallback)
    function injectTrackingIntoHtml(originalHtml, label) {
        if (!label || label.trim() === "") {
            throw new Error("Mall label is required. Please enter a unique kiosk location.");
        }
        const trackingBlock = getTrackingScript(label.trim());
        // Insert before  case-insensitive, also handle  with spaces
        const bodyCloseRegex = /<\/body\s*>/i;
        if (bodyCloseRegex.test(originalHtml)) {
            return originalHtml.replace(bodyCloseRegex, trackingBlock + '\n');
        }
        const htmlCloseRegex = /<\/html\s*>/i;
        if (htmlCloseRegex.test(originalHtml)) {
            return originalHtml.replace(htmlCloseRegex, trackingBlock + '\n');
        }
        // If no root tags, just append
        return originalHtml + '\n' + trackingBlock;
    }
